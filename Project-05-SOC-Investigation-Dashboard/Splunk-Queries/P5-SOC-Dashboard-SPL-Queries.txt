==================================================
01 — TOTAL SECURITY EVENTS
==================================================

index=main
| stats count AS Total_Events


==================================================
02 — FAILED LOGIN ATTEMPTS
==================================================

index=main sourcetype="XmlWinEventLog:Security"
| spath input=_raw output=EventID path=Event.System.EventID
| search EventID=4625
| stats count AS Failed_Login_Attempts


==================================================
03 — FAILED LOGIN ATTEMPTS OVER TIME
==================================================

index=main sourcetype="XmlWinEventLog:Security"
| spath input=_raw output=EventID path=Event.System.EventID
| search EventID=4625
| timechart span=1h count AS Failed_Login_Attempts


==================================================
04 — TOP 10 WINDOWS SECURITY EVENTS
==================================================

index=main sourcetype="XmlWinEventLog:Security"
| spath input=_raw output=EventID path=Event.System.EventID
| stats count by EventID
| sort -count
| head 10


==================================================
05 — FAILED LOGIN ATTEMPTS BY USER ACCOUNT
==================================================

index=main sourcetype="XmlWinEventLog:Security"
| spath input=_raw output=EventID path=Event.System.EventID
| search EventID=4625
| rex field=_raw "<Data Name='TargetUserName'>(?<Username>[^<]+)</Data>"
| stats count by Username
| sort -count
